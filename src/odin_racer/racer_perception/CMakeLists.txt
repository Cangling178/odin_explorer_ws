cmake_minimum_required(VERSION 3.8)
project(racer_perception)
find_package(ament_cmake REQUIRED)
install(FILES README.md README_cn.md DESTINATION share/${PROJECT_NAME})
foreach(asset_dir config launch urdf)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${asset_dir}")
    install(DIRECTORY ${asset_dir} DESTINATION share/${PROJECT_NAME})
  endif()
endforeach()
find_package(rclcpp REQUIRED)
find_package(racer_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
add_executable(line_perception src/line_perception.cpp)
target_compile_features(line_perception PRIVATE cxx_std_17)
target_compile_options(line_perception PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(line_perception PRIVATE include)
ament_target_dependencies(line_perception rclcpp racer_interfaces sensor_msgs nav_msgs std_msgs tf2 tf2_ros cv_bridge)
target_include_directories(line_perception PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(line_perception ${OpenCV_LIBRARIES})
add_executable(line_offline src/line_offline.cpp)
target_compile_features(line_offline PRIVATE cxx_std_17)
target_include_directories(line_offline PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(line_offline ${OpenCV_LIBRARIES})
install(TARGETS line_offline DESTINATION lib/${PROJECT_NAME})
install(TARGETS line_perception DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_line_geometry test/test_line_geometry.cpp)
  target_compile_features(test_line_geometry PRIVATE cxx_std_17)
  target_include_directories(test_line_geometry PRIVATE include)
  target_include_directories(test_line_geometry PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(test_line_geometry ${OpenCV_LIBRARIES})
endif()
ament_package()
