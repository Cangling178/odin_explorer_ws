cmake_minimum_required(VERSION 3.8)
project(racer_control)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
install(FILES README.md README_cn.md DESTINATION share/${PROJECT_NAME})
foreach(asset_dir config launch urdf)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${asset_dir}")
    install(DIRECTORY ${asset_dir} DESTINATION share/${PROJECT_NAME})
  endif()
endforeach()
find_package(rclcpp REQUIRED)
find_package(racer_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(line_controller src/line_controller.cpp)
target_compile_features(line_controller PRIVATE cxx_std_17)
target_compile_options(line_controller PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(line_controller PRIVATE include)
ament_target_dependencies(line_controller rclcpp racer_interfaces nav_msgs geometry_msgs std_msgs std_srvs tf2 tf2_ros)
install(TARGETS line_controller DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_lap_route test/test_lap_route.cpp)
  target_compile_features(test_lap_route PRIVATE cxx_std_17)
  target_include_directories(test_lap_route PRIVATE include)
  ament_add_gtest(test_line_tracker test/test_line_tracker.cpp)
  target_compile_features(test_line_tracker PRIVATE cxx_std_17)
  target_include_directories(test_line_tracker PRIVATE include)
endif()
find_package(Eigen3 REQUIRED)
find_package(racer_perception REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
add_executable(lap_controller src/lap_controller.cpp)
target_compile_features(lap_controller PRIVATE cxx_std_17)
target_compile_options(lap_controller PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(lap_controller PRIVATE include)
target_include_directories(lap_controller PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(lap_controller Eigen3::Eigen ${OpenCV_LIBRARIES})
ament_target_dependencies(lap_controller racer_perception sensor_msgs rclcpp racer_interfaces nav_msgs geometry_msgs std_msgs std_srvs tf2 tf2_ros)
install(TARGETS lap_controller DESTINATION lib/${PROJECT_NAME})
ament_package()
