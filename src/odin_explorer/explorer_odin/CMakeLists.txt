cmake_minimum_required(VERSION 3.8)
project(explorer_odin)
if(POLICY CMP0074)
  cmake_policy(SET CMP0074 NEW)
endif()
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters segmentation search)
add_executable(odin_nav_adapter src/odin_nav_adapter.cpp)
ament_target_dependencies(odin_nav_adapter rclcpp nav_msgs sensor_msgs tf2 tf2_geometry_msgs tf2_msgs tf2_ros pcl_conversions)
target_include_directories(odin_nav_adapter SYSTEM PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(odin_nav_adapter ${PCL_LIBRARIES})
if(BUILD_TESTING)
  enable_testing()
  add_executable(small_obstacle_filter_test test/small_obstacle_filter_test.cpp)
  target_include_directories(small_obstacle_filter_test SYSTEM PRIVATE ${PCL_INCLUDE_DIRS})
  target_link_libraries(small_obstacle_filter_test ${PCL_LIBRARIES})
  target_compile_options(small_obstacle_filter_test PRIVATE -UNDEBUG)
  add_test(NAME small_obstacle_filter COMMAND small_obstacle_filter_test)
endif()
install(TARGETS odin_nav_adapter DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
foreach(asset_dir config launch urdf meshes)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${asset_dir}")
    install(DIRECTORY ${asset_dir} DESTINATION share/${PROJECT_NAME})
  endif()
endforeach()
ament_package()
