cmake_minimum_required(VERSION 3.8)
project(explorer_localization)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(odin_cloud_gate src/odin_cloud_gate.cpp)
target_compile_features(odin_cloud_gate PUBLIC cxx_std_17)
target_compile_options(odin_cloud_gate PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(odin_cloud_gate rclcpp sensor_msgs std_msgs tf2 tf2_ros)
install(TARGETS odin_cloud_gate DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
foreach(asset_dir config launch urdf meshes)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${asset_dir}")
    install(DIRECTORY ${asset_dir} DESTINATION share/${PROJECT_NAME})
  endif()
endforeach()
ament_package()
